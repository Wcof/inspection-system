```typescript
// 检查周期配置Modal
const checkItemCycleModalVisible = ref(false)
const currentCheckItemCycle = ref({
  frequency: 'daily',
  cycleType: 'interval',
  cycleValue: 1
})
let currentEditingCheckItem: any = null

function openCheckItemCycleModal(record: any) {
  currentEditingCheckItem = record
  currentCheckItemCycle.value = {
    frequency: record.inspectionCycle?.frequency || 'daily',
    cycleType: record.inspectionCycle?.cycleType || 'interval',
    cycleValue: record.inspectionCycle?.cycleValue || 1
  }
  checkItemCycleModalVisible.value = true
}

function saveCheckItemCycle() {
  if (currentEditingCheckItem) {
    currentEditingCheckItem.inspectionCycle = { ...currentCheckItemCycle.value }
  }
  checkItemCycleModalVisible.value = false
}

function closeCheckItemCycleModal() {
  checkItemCycleModalVisible.value = false
  currentEditingCheckItem = null
}

// 检查窗口配置Modal
const checkItemWindowModalVisible = ref(false)
const currentCheckItemWindow = ref({
  timeRange: [null, null]
})

function openCheckItemWindowModal(record: any) {
  currentEditingCheckItem = record
  currentCheckItemWindow.value = {
    timeRange: record.inspectionWindow?.timeRange || [null, null]
  }
  checkItemWindowModalVisible.value = true
}

function saveCheckItemWindow() {
  if (currentEditingCheckItem) {
    currentEditingCheckItem.inspectionWindow = { ...currentCheckItemWindow.value }
  }
  checkItemWindowModalVisible.value = false
}

function closeCheckItemWindowModal() {
  checkItemWindowModalVisible.value = false
  currentEditingCheckItem = null
}
```
